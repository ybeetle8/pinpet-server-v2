# 添加执行权限
chmod +x nginx_setup.sh

# 交互式菜单（推荐）
sudo ./nginx_setup.sh

# 或直接命令行
sudo ./scripts/nginx_setup.sh install    # 安装
sudo ./scripts/nginx_setup.sh uninstall  # 卸载
sudo ./scripts/nginx_setup.sh status     # 查看状态
sudo ./scripts/nginx_setup.sh reload     # 重载配置


安装完成后执行加到开机启动:
systemctl start nginx && systemctl enable nginx
